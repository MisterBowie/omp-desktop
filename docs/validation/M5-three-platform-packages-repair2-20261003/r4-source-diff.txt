+function pnpmInvocation() {
+  const execPath = process.env.npm_execpath;
+  if (execPath && /pnpm/i.test(execPath)) {
+    return { command: process.execPath, prefix: [execPath], shell: false };
+  }
+  return {
+    command: process.platform === "win32" ? "pnpm.cmd" : "pnpm",
+    prefix: [],
